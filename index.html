Automação de Nome e Semana pelo App

O membro não precisa selecionar a semana nem digitar o nome no app. A planilha calcula a Semana do Mês automaticamente a partir da Data/Hora e busca o Nome do Colaborador na aba Base_Membros.

Código Apps Script Atualizado (GP Hidros):

function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("Registro_Presenca") || ss.getActiveSheet();
    var membrosSheet = ss.getSheetByName("Base_Membros");
    var data = JSON.parse(e.postData.contents);
    
    var email = (data.email || "").toLowerCase().trim();
    if (!email.endsWith("@hidrosconsultoria.com.br") && !email.endsWith("@gmail.com")) {
      return ContentService.createTextOutput(JSON.stringify({ 
        "status": "error", 
        "message": "E-mail não autorizado fora da Hidros." 
      })).setMimeType(ContentService.MimeType.JSON);
    }
    
    // Busca automática do nome na aba Base_Membros a partir do e-mail
    var nomeColaborador = "Não Cadastrado";
    if (membrosSheet && membrosSheet.getLastRow() > 4) {
      var dadosMembros = membrosSheet.getRange(5, 1, membrosSheet.getLastRow() - 4, 2).getValues();
      for (var i = 0; i < dadosMembros.length; i++) {
        var emailBase = (dadosMembros[i][0] || "").toString().toLowerCase().trim();
        if (emailBase === email) {
          nomeColaborador = dadosMembros[i][1];
          break;
        }
      }
    }

    // Cálculo da Semana do Mês (Semana 01, Semana 02, etc.)
    var now = new Date();
    var dataHoraStr = data.dataHora || now.toLocaleString("pt-BR");
    var dia = now.getDate();
    var primeiroDiaMes = new Date(now.getFullYear(), now.getMonth(), 1).getDay(); // 0 Dom, 1 Seg...
    var primeiroDiaAjustado = (primeiroDiaMes === 0 ? 6 : primeiroDiaMes - 1); // 0 Seg ... 6 Dom
    var semanaNum = Math.ceil((dia + primeiroDiaAjustado) / 7);
    var semanaTexto = (data.horarioPlantao === "RG" || data.tipo === "RG") ? "RG" : ("Semana " + (semanaNum < 10 ? "0" + semanaNum : semanaNum));
    
    // Cria cabeçalho automático se a planilha estiver vazia
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        "Data e Hora",
        "E-mail Institucional",
        "Colaborador",
        "Semana",
        "Dia da Semana",
        "Horário de Plantão",
        "Tipo de Ponto",
        "Atraso?",
        "Justificativa Oficial",
        "Observação"
      ]);
    }
    
    // Insere a nova linha com Nome e Semana gerados automaticamente
    sheet.appendRow([
      dataHoraStr,
      data.email,
      nomeColaborador,
      semanaTexto,
      data.diaSemana || "-",
      data.horarioPlantao || "-",
      data.tipoFormatado || data.tipo,
      data.isLate ? "SIM" : "NÃO",
      data.justificativa || "-",
      data.observacao || "-"
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({ 
      "status": "success", 
      "nome": nomeColaborador,
      "semana": semanaTexto
    })).setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ 
      "status": "error", 
      "message": error.toString() 
    })).setMimeType(ContentService.MimeType.JSON);
  }
}
